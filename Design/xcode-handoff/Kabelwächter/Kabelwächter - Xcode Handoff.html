Why a shared package?

iOS and tvOS share every token. The only platform delta is the type ramp (sofa distance ≈ 2.2× scale). Putting tokens in a Swift Package means one edit propagates to both apps — no copy-paste drift.

public extension KW.Font {
    // iOS / shared ramp
    static let display = Font.system(size: 56, weight: .bold,    design: .monospaced)
    static let title   = Font.system(size: 32, weight: .bold,    design: .monospaced)
    static let h2      = Font.system(size: 22, weight: .medium,  design: .monospaced)
    static let label   = Font.system(size: 11, weight: .medium,  design: .monospaced)
    static let telem   = Font.system(size: 13, weight: .regular, design: .monospaced)
    static let body    = Font.system(size: 15, weight: .regular)

    // tvOS ramp (sofa distance, ~2.2× scale)
    static let titleTV = Font.system(size: 76, weight: .bold,    design: .monospaced)
    static let labelTV = Font.system(size: 24, weight: .medium,  design: .monospaced)
    static let bodyTV  = Font.system(size: 32, weight: .regular)

    // Adaptive — picks the right scale per platform
    static var titleAdaptive: Font {
        #if os(tvOS)
        return titleTV
        #else
        return title
        #endif
    }
}

struct TunnelOverviewTV: View {
  @StateObject var vm = TunnelVM()
  @FocusState var focused: ID?

  var body: some View {
    ZStack {
      KWBackdrop()                  // grid + radial
      TunnelViz(state: vm.state)    // rings + rays

      VStack {
        HStack {                    // header
          Text("KABELWÄCHTER · v1.0.2")
            .kwLabel()
          Spacer()
          StatusPill(state: vm.state)
        }
        .padding(.top, KW.Space.safeTop)

        Spacer()

        VStack(spacing: KW.Space.md) {       // HUD
          Text("ACTIVE PEER").kwLabel()
          Text(vm.active.name)
            .font(KW.Font.titleTV)
          Text(vm.active.endpoint)
            .font(KW.Font.telemTV)
            .foregroundStyle(.kwTextDim)
        }

        Spacer()

        HStack(spacing: KW.Space.sm) {       // card row
          ForEach(vm.tunnels) { t in
            TunnelCard(tunnel: t,
                       active: t.id == vm.active.id)
              .focused($focused, equals: t.id)
              .kwFocusRing(focused == t.id)
          }
        }
      }
      .padding(.horizontal, KW.Space.gutter)
    }
    .background(Color.kwBg0)
  }
}

struct KWConnectButton: View {
    let state: KWConnectionState
    let action: () -> Void

    var body: some View {
        Button(action: action) {
            HStack(spacing: KW.Space.sm) {
                Image(systemName: state == .connected ? "bolt.slash.fill" : "bolt.fill")
                Text(state == .connected ? "DISCONNECT" : "CONNECT")
                    .tracking(2.2)
            }
            .font(KW.Font.labelAdaptive.weight(.semibold))
            .padding(.horizontal, KW.Space.lg)
            .padding(.vertical, KW.Space.md - 2)
            .foregroundStyle(Color.kwBg0)
            .background(state == .connected ? Color.kwError : Color.kwSignal)
            .shadow(color: (state == .connected ? Color.kwError : Color.kwSignal).opacity(0.45),
                    radius: KW.Glow.signal)
        }
        .buttonStyle(.plain)
        .animation(KW.Motion.snap, value: state)
    }
}

Xcode setup checklist

~10 minutes from clone to first preview
01

Create a new Xcode workspace. File → New → Workspace. Save as Kabelwaechter.xcworkspace.

02

Add a Multiplatform Swift Package called KabelwaechterUI (Library, no executable).

03

Copy xcode-handoff/KabelwaechterUI/* into Sources/KabelwaechterUI/. Drag Colors.xcassets into the same folder.

04

In Package.swift, list the asset catalog as a resource:

targets: [
  .target(
    name: "KabelwaechterUI",
    resources: [.process("Colors.xcassets")]
  )
]
05

Add two app targets: Kabelwaechter-iOS and Kabelwaechter-tvOS. Both depend on KabelwaechterUI.

06

In each app target's Info.plist, set UIUserInterfaceStyle = Dark. The brand is dark-only.

07

Drop the SVG icon files into AppIcon.appiconset (iOS) and a tvAppIcon.brandassets with three imagestacklayer folders (tvOS).

08

Verify with a smoke test:

import SwiftUI
import KabelwaechterUI

struct ContentView: View {

      Text("[ KABELWÄCHTER ]").kwLabel()

Things you do NOT need
  • Custom font files — SF Mono ships with every Apple OS via design: .monospaced.
  • A separate codebase per platform — every token, modifier, and primitive is platform-agnostic. Only the type ramp branches on #if os(tvOS).
  • Third-party theming libraries — the whole token layer is < 300 lines of plain Swift.
